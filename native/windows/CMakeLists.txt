cmake_minimum_required(VERSION 3.20)
project(titen_audio LANGUAGES CXX)
add_executable(titen-audio capture.cpp)
target_compile_features(titen-audio PRIVATE cxx_std_17)
if(WIN32)
  target_compile_definitions(titen-audio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(titen-audio PRIVATE ole32 uuid propsys)
  if(MINGW)
    target_link_libraries(titen-audio PRIVATE ksguid)
  endif()
  if(MSVC)
    target_compile_options(titen-audio PRIVATE /W4 /EHsc)
    set_property(TARGET titen-audio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_link_options(titen-audio PRIVATE -static -municode)
  endif()
endif()
enable_testing()
add_test(NAME audio-protocol COMMAND titen-audio --self-test)
